rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // — Helpers

    function isSignedIn() {
      return request.auth != null;
    }

    function isPlatformAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/platform_admins/$(request.auth.uid));
    }

    function tokenTenantId() {
      return request.auth.token.tenantId;
    }

    function isTenantMember(tenantId) {
      return isSignedIn() && tokenTenantId() == tenantId;
    }

    function isTenantOwner(tenantId) {
      return isTenantMember(tenantId) && request.auth.token.role == "owner";
    }

    function isGestor(tenantId) {
      return isTenantMember(tenantId) && request.auth.token.get('role', '') == 'staff_bookings';
    }

    function isEquipaClinica(tenantId) {
      return isTenantMember(tenantId) && request.auth.token.get('role', '') != 'staff_bookings';
    }

    function gestorPode(tenantId, permissao) {
      return isGestor(tenantId) &&
        get(/databases/$(database)/documents/tenants/$(tenantId)/staff/$(request.auth.uid)).data.get('permissoes', {}).get(permissao, false) == true;
    }

    function tenantIsActive(tenantId) {
      let t = get(/databases/$(database)/documents/tenants/$(tenantId)).data;
      return t.status == "active" ||
        (t.status == "trial" && (!("trialEndsAt" in t) || request.time < t.trialEndsAt));
    }

    match /platform_admins/{uid} {
      allow read: if isSignedIn() && request.auth.uid == uid;
      allow write: if false;
    }

    match /settings/{settingId} {
      allow read: if isSignedIn() || settingId == 'platformConfig';
      allow write: if isPlatformAdmin();
    }

    match /tenants/{tenantId} {
      allow read: if isTenantMember(tenantId) || isPlatformAdmin();

      allow create: if isPlatformAdmin() ||
        (isSignedIn() &&
        request.resource.data.ownerUid == request.auth.uid &&
        request.resource.data.status == "pending_activation");

      // O dono pode editar os seus dados, mas NÃO os campos controlados pela
      // plataforma (estado, plano, módulos exclusivos, acessos e limites extra,
      // datas de teste/pagamento, notas internas).
      allow update: if (isTenantOwner(tenantId) &&
        request.resource.data.status == resource.data.status &&
        request.resource.data.plan == resource.data.plan &&
        !request.resource.data.diff(resource.data).affectedKeys().hasAny([
          'modulos', 'fullAccessOverride', 'limitOverrides', 'trialEndsAt',
          'nextPaymentDue', 'adminNotes', 'statusChangedAt', 'ownerUid'
        ]))
        || isPlatformAdmin();

      allow delete: if isPlatformAdmin();

      match /staff/{uid} {
        allow read: if isTenantMember(tenantId);
        allow write: if isTenantOwner(tenantId) || isPlatformAdmin();
      }

      match /staffInvites/{inviteId} {
        allow read: if isTenantOwner(tenantId) || isPlatformAdmin();
        allow create, delete: if isTenantOwner(tenantId) || isPlatformAdmin();
        allow update: if false;
      }

      match /services/{serviceId} {
        allow read: if true;
        allow write: if isTenantOwner(tenantId) && tenantIsActive(tenantId);
      }

      match /serviceCategories/{categoryId} {
        allow read: if true;
        allow write: if isTenantOwner(tenantId) && tenantIsActive(tenantId);
      }

      match /patients/{patientId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow delete: if isEquipaClinica(tenantId);
        allow create: if gestorPode(tenantId, 'criarPacientes') && tenantIsActive(tenantId) &&
          request.resource.data.keys().hasOnly(['fullName', 'phone', 'email', 'createdAt', 'criadoPorGestor', 'active']);

        match /reports/{reportId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }

        match /formResponses/{responseId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }

        match /messages/{messageId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update: if isEquipaClinica(tenantId) && tenantIsActive(tenantId) &&
            request.resource.data.diff(resource.data).affectedKeys().hasOnly(['readByProfessional', 'readByPatient']);
          allow delete: if false;
        }

        match /checkins/{checkinId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }

        match /healingProtocols/{protocolId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }
      }

      match /contentVersions/{versaoId} {
        allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow update: if false;
        allow delete: if isEquipaClinica(tenantId);
      }

      match /mapaAlmaLeads/{leadId} {
        allow create: if request.resource.data.nome is string &&
          request.resource.data.dataNascimento is string &&
          request.resource.data.paymentStatus == 'unpaid';
        allow read: if isEquipaClinica(tenantId) || isPlatformAdmin();
        allow update: if isTenantOwner(tenantId) &&
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['paymentStatus','paidAt']);
      }

      match /marketingContacts/{contactId} {
        allow create, update: if request.resource.data.nome is string &&
          request.resource.data.contacto is string;
        allow read: if isEquipaClinica(tenantId) || isPlatformAdmin();
        allow delete: if isTenantOwner(tenantId) || isPlatformAdmin();
      }

      match /cartomanciaConsultas/{consultaId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow delete: if isEquipaClinica(tenantId);
      }

      match /cartomanciaLeads/{leituraId} {
        allow create: if request.resource.data.nome is string &&
                        request.resource.data.contacto is string;
        allow read, delete: if isEquipaClinica(tenantId) || isPlatformAdmin();
      }

      match /triagemBaixaVibracao/{triagemId} {
        allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow update, delete: if false;
      }

      match /reikiConfig/{configId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiStudents/{studentId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow delete: if isEquipaClinica(tenantId);

        match /certificados/{certId} {
          allow read, create: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }
      }

      match /reikiLibrary/{recursoId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiCategories/{categoriaId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiModules/{moduloId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiClasses/{turmaId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiQuizzes/{testeId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /reikiQuizResults/{resultadoId} {
        allow create: if request.resource.data.codigo is string &&
                        request.resource.data.testeId is string;
        allow read: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow update, delete: if false;
      }

      match /reikiThreads/{codigo} {
        allow get: if true;
        allow list: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow write: if true;

        match /mensagens/{msgId} {
          allow read: if true;
          allow create: if request.resource.data.texto is string &&
                          request.resource.data.texto.size() > 0 &&
                          request.resource.data.texto.size() < 4000;
          allow update: if false;
          allow delete: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        }
      }

      match /reikiChat/{sala} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);

        match /mensagens/{msgId} {
          allow read: if true;
          allow create: if request.resource.data.texto is string &&
                          request.resource.data.texto.size() > 0 &&
                          request.resource.data.texto.size() < 4000 &&
                          request.resource.data.nome is string;
          allow update: if false;
          allow delete: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        }
      }

      match /reikiAccess/{codigo} {
        allow get: if true;
        allow list: if false;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /protocols/{protocolId} {
        allow read, write: if (isEquipaClinica(tenantId) && tenantIsActive(tenantId)) || isPlatformAdmin();
      }

      match /forms/{formId} {
        allow read: if true;
        allow write: if (isEquipaClinica(tenantId) && tenantIsActive(tenantId)) || isPlatformAdmin();
      }

      match /teamMembers/{memberId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /trainings/{trainingId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /testimonials/{testimonialId} {
        allow read: if true;
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /patientContacts/{patientId} {
        allow read: if isTenantMember(tenantId) && tenantIsActive(tenantId);
        allow write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow create: if gestorPode(tenantId, 'criarPacientes') && tenantIsActive(tenantId) &&
          request.resource.data.keys().hasOnly(['fullName', 'phone', 'email', 'updatedAt']);
      }

      match /appointments/{appointmentId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow read, create: if isGestor(tenantId) && tenantIsActive(tenantId);
        allow update: if isGestor(tenantId) && tenantIsActive(tenantId) && (
          gestorPode(tenantId, 'remarcar') ||
          (gestorPode(tenantId, 'cancelar') &&
            request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']) &&
            request.resource.data.status == 'cancelled') ||
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['googleEventId', 'googleSyncStatus', 'googleSyncError'])
        );
        allow delete: if gestorPode(tenantId, 'cancelar') && tenantIsActive(tenantId);
      }

      match /waitlist/{waitlistId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
        allow read: if isGestor(tenantId) && tenantIsActive(tenantId);
        allow write: if gestorPode(tenantId, 'listaEspera') && tenantIsActive(tenantId);
      }

      match /knowledgeBase/{entryId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /farmacia/{entryId} {
        allow read, write: if isEquipaClinica(tenantId) && tenantIsActive(tenantId);
      }

      match /auditLog/{logId} {
        allow read: if isTenantOwner(tenantId) || isPlatformAdmin();
        allow create: if isTenantMember(tenantId);
        allow update, delete: if false;
      }
    }

    match /platformProtocolTemplates/{templateId} {
      allow read, write: if isPlatformAdmin();
    }

    match /platformFormTemplates/{templateId} {
      allow read, write: if isPlatformAdmin();
    }

    match /privateMethodRequests/{requestId} {
      allow create: if isSignedIn() &&
        request.resource.data.tenantId == request.auth.token.tenantId;
      allow read: if isPlatformAdmin() || (isSignedIn() && resource.data.tenantId == request.auth.token.tenantId);
      allow update, delete: if isPlatformAdmin();
    }

    match /inviteCodes/{code} {
      allow read: if false;
      allow create: if isSignedIn() &&
        request.resource.data.tenantId == request.auth.token.tenantId &&
        request.auth.token.role == "owner";
      allow update, delete: if isSignedIn() &&
        resource.data.tenantId == request.auth.token.tenantId;
    }

    match /public_tenant_profiles/{profileId} {
      allow read: if true;
      allow write: if isSignedIn() && request.auth.token.get('role', '') != 'staff_bookings' &&
        request.resource.data.tenantId == request.auth.token.tenantId &&
        (resource == null || resource.data.tenantId == request.auth.token.tenantId);
    }

    match /public_booking_requests/{requestId} {
      allow create: if request.resource.data.tenantId is string &&
                      request.resource.data.status == "pending_review";
      allow read, update: if isSignedIn() &&
        resource.data.tenantId == request.auth.token.tenantId &&
        (request.auth.token.get('role', '') != 'staff_bookings' ||
         gestorPode(resource.data.tenantId, 'pedidos'));
      allow delete: if false;
    }

    match /public_form_responses/{responseId} {
      // Terapeuta cria o documento e o paciente recebe o link
      allow create: if request.resource.data.tenantId is string &&
                      request.resource.data.formId is string &&
                      request.resource.data.status == "pending_review";

      // Paciente lê (sem autenticação) enquanto a ficha está por preencher
      allow read: if resource.data.status == 'pending_review'
                  || (isSignedIn() &&
                      resource.data.tenantId == request.auth.token.tenantId &&
                      request.auth.token.get('role', '') != 'staff_bookings');

      // Paciente submete as respostas (sem autenticação):
      //   • só quando a ficha ainda está pendente
      //   • apenas pode alterar respostas, nome e data de conclusão
      //   • o status não pode ser alterado pelo paciente (fica pending_review)
      allow update: if resource.data.status == 'pending_review' &&
                      request.resource.data.status == 'pending_review' &&
                      request.resource.data.diff(resource.data)
                        .affectedKeys()
                        .hasOnly(['respostas', 'respondentName', 'completadoEm'])
                    || (isSignedIn() &&
                        resource.data.tenantId == request.auth.token.tenantId &&
                        request.auth.token.get('role', '') != 'staff_bookings');

      // RGPD: a equipa clínica do próprio tenant pode apagar respostas
      allow delete: if isSignedIn() &&
        resource.data.tenantId == request.auth.token.tenantId &&
        request.auth.token.get('role', '') != 'staff_bookings';
    }

    match /public_training_signups/{signupId} {
      allow create: if request.resource.data.tenantId is string &&
                      request.resource.data.trainingId is string;
      allow read, update: if isSignedIn() &&
        resource.data.tenantId == request.auth.token.tenantId &&
        request.auth.token.get('role', '') != 'staff_bookings';
      allow delete: if false;
    }

    match /platform_support_requests/{requestId} {
      allow create: if isSignedIn() &&
        request.resource.data.tenantId == request.auth.token.tenantId;
      allow read: if isPlatformAdmin() ||
        (isSignedIn() && resource.data.tenantId == request.auth.token.tenantId);
      allow update: if isPlatformAdmin();
      allow delete: if false;
    }

    // ── Módulo: Pesquisa Emocional Clínica ───────────────────────────────
    match /modulo_requests/{requestId} {
      allow read: if isSignedIn() &&
        resource.data.tenantId == request.auth.token.tenantId;
      allow write: if false;
    }

    match /pe_questionarios/{qId} {
      allow read: if resource.data.status == 'pendente'
                  || (isSignedIn() && resource.data.tenantId == request.auth.token.tenantId);
      allow create: if isSignedIn()
                    && request.resource.data.tenantId == request.auth.token.tenantId;
      allow update: if false;
      allow delete: if false;
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
