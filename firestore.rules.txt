rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ──────────────────────────────────────────────────────────
    function isSignedIn() {
      return request.auth != null;
    }

    function isPlatformAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/platform_admins/$(request.auth.uid));
    }

    // O claim tenantId vem do token (custom claim), nunca de dados enviados
    // pelo cliente. Isto é o que impede um subscritor de "fingir" pertencer
    // a outro tenant.
    function tokenTenantId() {
      return request.auth.token.tenantId;
    }

    function isTenantMember(tenantId) {
      return isSignedIn() && tokenTenantId() == tenantId;
    }

    function isTenantOwner(tenantId) {
      return isTenantMember(tenantId) && request.auth.token.role == "owner";
    }

    function tenantIsActive(tenantId) {
      return get(/databases/$(database)/documents/tenants/$(tenantId)).data.status
        in ["trial", "active"];
    }

    // ── Plataforma (só tu) ──────────────────────────────────────────────
    match /platform_admins/{uid} {
      allow read: if isSignedIn() && request.auth.uid == uid;
      allow write: if false; // só via Admin SDK / consola, nunca do cliente
    }

    // ── Tenants ─────────────────────────────────────────────────────────
    match /tenants/{tenantId} {
      // Leitura pública limitada (para a página de marcações) é feita via
      // uma coleção separada /public_tenant_profiles/{tenantId}, NUNCA aqui.
      allow read: if isTenantMember(tenantId) || isPlatformAdmin();

      // Criação do tenant só acontece via Cloud Function (onboarding
      // controlado), nunca diretamente do cliente.
      allow create: if false;

      // Só o owner do tenant pode editar a própria configuração; campos
      // sensíveis (status, plan) só a plataforma pode alterar.
      allow update: if isTenantOwner(tenantId) &&
        request.resource.data.status == resource.data.status &&
        request.resource.data.plan == resource.data.plan
        || isPlatformAdmin();

      allow delete: if isPlatformAdmin();

      // ── Staff ──
      match /staff/{uid} {
        allow read: if isTenantMember(tenantId);
        allow write: if isTenantOwner(tenantId) || isPlatformAdmin();
      }

      // ── Serviços/catálogo ──
      match /services/{serviceId} {
        allow read: if isTenantMember(tenantId);
        allow write: if isTenantOwner(tenantId) && tenantIsActive(tenantId);
      }

      // ── Pacientes (categoria especial RGPD — regras mais restritas) ──
      match /patients/{patientId} {
        allow read, write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
        // Apagar um paciente é permitido (direito ao apagamento), mas fica
        // sempre registado no auditLog pela aplicação antes de chamar delete.
        allow delete: if isTenantMember(tenantId);
      }

      // ── Marcações ──
      match /appointments/{appointmentId} {
        allow read, write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // ── Audit log — nunca editável ou apagável, só criado pela app ──
      match /auditLog/{logId} {
        allow read: if isTenantOwner(tenantId) || isPlatformAdmin();
        allow create: if isTenantMember(tenantId);
        allow update, delete: if false; // imutável — garante valor como prova
      }
    }

    // ── Perfis públicos (para a página de marcações do subscritor) ──────
    // Coleção separada, deliberadamente: contém só o que pode ser público
    // (nome, branding, serviços, horários), nunca pacientes/marcações.
    match /public_tenant_profiles/{tenantId} {
      allow read: if true;
      allow write: if false; // gerido só por Cloud Function, espelha o tenant
    }

    // ── Marcações públicas criadas por clientes finais no funil ─────────
    // O cliente final não tem conta — a marcação entra como "pending" numa
    // coleção separada e uma Cloud Function valida/promove para
    // /tenants/{tenantId}/appointments, evitando escrita direta não
    // autenticada na coleção principal.
    match /public_booking_requests/{requestId} {
      allow create: if request.resource.data.tenantId is string &&
                     request.resource.data.status == "pending_review";
      allow read, update, delete: if false; // só Cloud Function processa
    }

    // ── Bloqueio total por omissão ───────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
