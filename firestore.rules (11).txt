rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ──────────────────────────────────────────────────────────
    function isSignedIn() {
      return request.auth != null;
    }

    function isPlatformAdmin() {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/platform_admins/$(request.auth.uid));
    }

    // O claim tenantId vem do token (custom claim), nunca de dados enviados
    // pelo cliente. Isto é o que impede um subscritor de "fingir" pertencer
    // a outro tenant.
    function tokenTenantId() {
      return request.auth.token.tenantId;
    }

    function isTenantMember(tenantId) {
      return isSignedIn() && tokenTenantId() == tenantId;
    }

    function isTenantOwner(tenantId) {
      return isTenantMember(tenantId) && request.auth.token.role == "owner";
    }

    function tenantIsActive(tenantId) {
      let t = get(/databases/$(database)/documents/tenants/$(tenantId)).data;
      return t.status == "active" ||
        (t.status == "trial" && (!("trialEndsAt" in t) || request.time < t.trialEndsAt));
    }

    // ── Plataforma (só tu) ──────────────────────────────────────────────
    match /platform_admins/{uid} {
      allow read: if isSignedIn() && request.auth.uid == uid;
      allow write: if false; // só via Admin SDK / consola, nunca do cliente
    }

    // ── Tenants ─────────────────────────────────────────────────────────
    match /tenants/{tenantId} {
      // Leitura pública limitada (para a página de marcações) é feita via
      // uma coleção separada /public_tenant_profiles/{tenantId}, NUNCA aqui.
      allow read: if isTenantMember(tenantId) || isPlatformAdmin();

      // Criação do tenant — só a plataforma pode criar diretamente pelo
      // painel de administração (a conta do subscritor continua a ter de
      // ser criada à parte, na consola Firebase, até termos Cloud
      // Functions ativas para automatizar isso também).
      // Criação do tenant: a plataforma cria com qualquer estado; um
      // subscritor pode registar-se a si próprio, mas o registo entra
      // sempre em "pending_activation" — nunca em estado já ativo — até
      // a plataforma validar a conta e mudar o estado manualmente.
      allow create: if isPlatformAdmin() ||
        (isSignedIn() &&
         request.resource.data.ownerUid == request.auth.uid &&
         request.resource.data.status == "pending_activation");

      // Só o owner do tenant pode editar a própria configuração; campos
      // sensíveis (status, plan) só a plataforma pode alterar.
      allow update: if isTenantOwner(tenantId) &&
        request.resource.data.status == resource.data.status &&
        request.resource.data.plan == resource.data.plan
        || isPlatformAdmin();

      allow delete: if isPlatformAdmin();

      // ── Staff ──
      match /staff/{uid} {
        allow read: if isTenantMember(tenantId);
        allow write: if isTenantOwner(tenantId) || isPlatformAdmin();
      }

      // ── Serviços/catálogo ──
      match /services/{serviceId} {
        allow read: if true;
        allow write: if isTenantOwner(tenantId) && tenantIsActive(tenantId);
      }

      // ── Pacientes (categoria especial RGPD — regras mais restritas) ──
      match /patients/{patientId} {
        allow read, write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
        // Apagar um paciente é permitido (direito ao apagamento), mas fica
        // sempre registado no auditLog pela aplicação antes de chamar delete.
        allow delete: if isTenantMember(tenantId);

        // Relatórios gerados por consulta guiada — dados de saúde,
        // mesma proteção que a ficha do paciente.
        match /reports/{reportId} {
          allow read, create: if isTenantMember(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false; // relatório gerado é imutável
        }

        // Respostas de fichas já associadas a este paciente.
        match /formResponses/{responseId} {
          allow read, create: if isTenantMember(tenantId) && tenantIsActive(tenantId);
          allow update, delete: if false;
        }
      }

      // Protocolos / métodos de atendimento — propriedade intelectual do
      // subscritor (o "know-how" da terapia dele). Nunca público.
      match /protocols/{protocolId} {
        allow read, write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // Fichas/questionários — leitura pública necessária para o paciente
      // conseguir preencher sem ter conta. Só a estrutura das perguntas é
      // pública, nunca as respostas.
      match /forms/{formId} {
        allow read: if true;
        allow write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // Equipa mostrada publicamente — não são contas de login.
      match /teamMembers/{memberId} {
        allow read: if true;
        allow write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // Formações/eventos — estrutura pública, inscrições ficam à parte.
      match /trainings/{trainingId} {
        allow read: if true;
        allow write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // Testemunhos — o subscritor escreve-os (a mão, ou copiados de redes
      // sociais); mostrados publicamente na página inicial.
      match /testimonials/{testimonialId} {
        allow read: if true;
        allow write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // ── Marcações ──
      match /appointments/{appointmentId} {
        allow read, write: if isTenantMember(tenantId) && tenantIsActive(tenantId);
      }

      // ── Audit log — nunca editável ou apagável, só criado pela app ──
      match /auditLog/{logId} {
        allow read: if isTenantOwner(tenantId) || isPlatformAdmin();
        allow create: if isTenantMember(tenantId);
        allow update, delete: if false; // imutável — garante valor como prova
      }
    }

    // ── Perfis públicos (para a página inicial e o funil de marcações) ──
    // Antes só uma Cloud Function podia escrever aqui — como ainda não
    // construímos essa função, o próprio subscritor pode escrever no seu
    // perfil público, mas só nele: o tenantId gravado tem de ser
    // exatamente o do seu próprio token, nunca de outro subscritor.
    match /public_tenant_profiles/{profileId} {
      allow read: if true;
      allow write: if isSignedIn() &&
        request.resource.data.tenantId == request.auth.token.tenantId &&
        (resource == null || resource.data.tenantId == request.auth.token.tenantId);
    }

    // ── Marcações públicas criadas por clientes finais no funil ─────────
    // O cliente final não tem conta — a marcação entra como "pending" numa
    // coleção separada. Como ainda não há Cloud Function ativa, o próprio
    // subscritor (autenticado, tenantId a bater certo) processa-a
    // manualmente no painel — mesmo padrão já usado nas Fichas e Formações.
    match /public_booking_requests/{requestId} {
      allow create: if request.resource.data.tenantId is string &&
                     request.resource.data.status == "pending_review";
      allow read, update: if isSignedIn() && resource.data.tenantId == request.auth.token.tenantId;
      allow delete: if false;
    }

    // ── Respostas de fichas públicas ──────────────────────────────────
    // O paciente não tem conta, por isso a resposta entra numa fila. O
    // subscritor (autenticado, com o próprio tenantId no token) pode ler e
    // marcar como processada as respostas que pertencem ao seu tenant —
    // isto evita depender de uma Cloud Function só para isto.
    match /public_form_responses/{responseId} {
      allow create: if request.resource.data.tenantId is string &&
                     request.resource.data.formId is string &&
                     request.resource.data.status == "pending_review";
      allow read, update: if isSignedIn() && resource.data.tenantId == request.auth.token.tenantId;
      allow delete: if false;
    }

    // ── Inscrições em formações/eventos ────────────────────────────────
    // Mesmo padrão de segurança das fichas: sem conta, entra numa fila
    // que só o subscritor dono desse tenant pode ler/atualizar.
    match /public_training_signups/{signupId} {
      allow create: if request.resource.data.tenantId is string &&
                     request.resource.data.trainingId is string;
      allow read, update: if isSignedIn() && resource.data.tenantId == request.auth.token.tenantId;
      allow delete: if false;
    }

    // ── Bloqueio total por omissão ───────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
